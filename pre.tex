\section{Prior Work and Gap}

\begin{frame}{Prior Work}
  \begin{itemize}
        \item CBF/HOCBF methods provide \textbf{formal safety guarantees}.
        \item Differentiable QP layers allow \textbf{end-to-end training}.
        \item Adaptive CBFs reduce conservatism in principle.
  \end{itemize}
\end{frame}

\begin{frame}{Research Gap}

\textit{Trainable, interpretable, environment-dependent, and still mathematically safe.}


\begin{alertblock}{What is still missing}
    \begin{itemize}
    \item Standard CBF parameters are usually \textbf{fixed}.
    \item Adaptive CBFs need \textbf{auxiliary dynamics} that are hard to design.
    \item Safety filters such as DFB are safe, but can remain \textbf{overly conservative}.
    \end{itemize}
\end{alertblock}


\end{frame}


\section{Methods and Procedures}

\begin{frame}{BarrierNet Architecture}
\centering
    \includegraphics[width=\linewidth,height=0.64\textheight,keepaspectratio]{fig1}

    \begin{columns}[T,onlytextwidth]

  \begin{column}{0.33\textwidth}
    \textbf{Input:}
    \begin{itemize}
      \item robot state $x$
      \item observation / environment feature $z$
    \end{itemize}


  \end{column}
  \begin{column}{0.33\textwidth}
    \textbf{Inside BarrierNet:}
    \begin{itemize}
      \item trainable penalty functions $p_i(z)$
      \item differentiable QP layer
      \item HOCBF constraints
    \end{itemize}    
  \end{column}
  \begin{column}{0.33\textwidth}
    \textbf{Output:} safe action $u^\ast$ with collision-free guarantee.    
  \end{column}
\end{columns}
\end{frame}

\begin{frame}[shrink=2]{Mathematical Formulation}
\footnotesize
BarrierNet replaces fixed HOCBF coefficients with penalties that \textbf{depend on the environment}:
\[
\psi_i(x,z)=\dot{\psi}_{i-1}(x,z)+p_i(z)\alpha_i\!\left(\psi_{i-1}(x,z)\right)
\]

\vspace{2pt}
The final safe action is produced by a differentiable quadratic program:
\[
u^\ast=\arg\min_u \frac{1}{2}u^\top H(z)u + F(z)^\top u
\]
subject to softened HOCBF constraints and control bounds.

\vspace{2pt}
\begin{itemize}
  \item $p_i(z) > 0$ are learned from data, so the safety margin changes with the scene.
  \item The QP stays differentiable, so the whole controller can still be trained end to end.
  \item BarrierNet learns \textbf{when to be stricter and when not to be overly conservative}.
\end{itemize}
\end{frame}

\begin{frame}{Discussion}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{exampleblock}{Main strengths}
      \begin{itemize}
        \item Clear integration of \textbf{control theory} and \textbf{deep learning}.
        \item Safety is theoretically justified, not only reported by experiments.
        \item Trainable penalty functions provide an interpretable mechanism for adaptivity.
        \item Runtime is realistic for online use.
      \end{itemize}
    \end{exampleblock}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{alertblock}{Main limitations}
      \begin{itemize}
        \item It assumes \textbf{known system dynamics} and user-defined safe sets.
        \item The number of constraints is fixed in advance.
        \item Evidence is mostly from simulation, with limited baseline diversity.
        \item Safety near the boundary still depends on discretization effects.
      \end{itemize}
    \end{alertblock}
  \end{column}
\end{columns}

\vspace{4pt}
\small
\textit{Overall judgement: the paper is convincing as a principled safety layer, but it is not yet a full solution to uncertainty, perception error, or real-world deployment.}
\end{frame}
